## common.py — 公共配置与工具函数

```python
"""第三章 动态图表复现 · 公共配置与工具函数。

- PALETTE：与 Excel 默认主题接近的商务配色
- save_html：渲染 pyecharts 图表为可交互 HTML（输出到仓库根目录 output/）
- snapshot：用 Playwright 把 HTML 截取为 PNG（输出到仓库根目录 images/），
  未安装 Playwright 时自动跳过，不影响 HTML 生成
"""
import os

# 与 Excel 默认主题接近的商务配色
PALETTE = ["#4472C4", "#ED7D31", "#A5A5A5", "#FFC000", "#5B9BD5", "#70AD47"]

REPO_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DATA_DIR = os.path.join(REPO_ROOT, "data")
OUTPUT_DIR = os.path.join(REPO_ROOT, "output")
IMAGES_DIR = os.path.join(REPO_ROOT, "images")


def save_html(chart, name):
    """渲染 pyecharts 图表为可交互的 HTML 文件，返回输出路径。"""
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    path = os.path.join(OUTPUT_DIR, f"{name}.html")
    chart.render(path)
    print(f"[HTML] {path}")
    return path


def snapshot(name, width=1100, height=760, wait=2000):
    """用 Playwright 把 output/{name}.html 截取为 images/{name}.png。

    HTML 中的图表依赖在线 JS 资源（assets.pyecharts.org），截图需要联网。
    未安装 Playwright 时打印提示并跳过，不视为错误。
    """
    html_path = os.path.join(OUTPUT_DIR, f"{name}.html")
    png_path = os.path.join(IMAGES_DIR, f"{name}.png")
    try:
        from playwright.sync_api import sync_playwright
    except ImportError:
        print(f"[跳过截图] 未安装 playwright：{png_path} 未生成（HTML 不受影响）")
        return None
    os.makedirs(IMAGES_DIR, exist_ok=True)
    with sync_playwright() as p:
        browser = p.chromium.launch(
            executable_path="/usr/bin/chromium" if os.path.exists("/usr/bin/chromium") else None,
            args=["--no-sandbox", "--disable-gpu"],
        )
        page = browser.new_page(viewport={"width": width, "height": height})
        page.goto("file://" + html_path)
        page.wait_for_timeout(wait)
        page.screenshot(path=png_path, full_page=True)
        browser.close()
    print(f"[PNG ] {png_path}")
    return png_path
```


## 01_动态柱形图.py — 01_动态柱形图

```python
"""01 动态柱形图：各区域销售额（Bar + Timeline 时间轴）

Excel 原效果：插入滚动条控件，拖动切换月份。
Python 复现：Bar 按月份逐张生成，Timeline 时间轴串联，
            拖动时间轴即可播放 1—6 月各区域销售额。
数据：各区域月度销售额（单位与教材一致）
"""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common import PALETTE, save_html, snapshot

from pyecharts import options as opts
from pyecharts.charts import Bar, Timeline
from pyecharts.commons.utils import JsCode

MONTHS = ["1月", "2月", "3月", "4月", "5月", "6月"]
REGIONS = ["华北", "华南", "东北", "西北", "西南", "华东"]
SALES = {
    "1月": [259, 247, 705, 513, 463, 384],
    "2月": [235, 266, 529, 324, 579, 282],
    "3月": [330, 342, 705, 540, 492, 308],
    "4月": [282, 342, 599, 270, 492, 384],
    "5月": [282, 323, 705, 486, 405, 308],
    "6月": [965, 380, 282, 567, 463, 897],
}


def build_chart():
    tl = Timeline(init_opts=opts.InitOpts(width="920px", height="520px"))
    for m in MONTHS:
        bar = (
            Bar()
            .add_xaxis(REGIONS)
            .add_yaxis(
                "销售额", SALES[m],
                label_opts=opts.LabelOpts(position="top", font_size=13),
                itemstyle_opts=opts.ItemStyleOpts(color=JsCode(
                    "function(p){var cs=" + str(PALETTE) + ";return cs[p.dataIndex % cs.length];}")),
            )
            .set_global_opts(
                title_opts=opts.TitleOpts(title=f"{m}各区域销售额", subtitle="拖动下方时间轴切换月份"),
                yaxis_opts=opts.AxisOpts(name="销售额", max_=1000),
                legend_opts=opts.LegendOpts(is_show=False),
            )
        )
        tl.add(bar, m)
    tl.add_schema(play_interval=1500, is_auto_play=False, is_loop_play=False,
                  pos_bottom="0%", control_position="left")
    return tl


if __name__ == "__main__":
    save_html(build_chart(), "01_动态柱形图")
    snapshot("01_动态柱形图")
```

## 02_动态跑道图.py — 02_动态跑道图

```python
"""02 动态跑道图：各部门人数（多环嵌套 Pie + Timeline 时间轴）

Excel 原效果：数据按部门呈环形"跑道"排布，组合框切换月份。
Python 复现：每个部门一个圆环（数值段 + 灰色占位段），
            中心文字显示当月公司总人数，Timeline 切换月份。
数据：各部门月度人数（单位：人）
"""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common import PALETTE, save_html, snapshot

from pyecharts import options as opts
from pyecharts.charts import Pie, Timeline
from pyecharts.commons.utils import JsCode

MONTHS = ["1月", "2月", "3月", "4月"]
DEPTS = ["人力部", "行政部", "财务部", "工程部", "采购部", "销售部"]
HEADCOUNT = {
    "1月": [130, 226, 238, 293, 326, 451],
    "2月": [138, 206, 228, 305, 349, 456],
    "3月": [130, 217, 255, 314, 316, 406],
    "4月": [129, 226, 226, 264, 359, 433],
}

# 只给"数值段"显示标签，占位段隐藏
LABEL_ONLY_VALUE = JsCode(
    "function(p){return p.name === '占位' ? '' : p.name + '\\n' + p.value + '人';}")


def build_chart():
    tl = Timeline(init_opts=opts.InitOpts(width="920px", height="600px"))
    for m in MONTHS:
        total = sum(HEADCOUNT[m])                    # 当月公司总人数
        pie = Pie(init_opts=opts.InitOpts(width="920px", height="600px"))
        for i, dept in enumerate(DEPTS):
            v = HEADCOUNT[m][i]
            placeholder = total / 2 - v              # 占位 = 总人数/2 − 部门人数（与 Excel 一致）
            inner, outer = 18 + i * 12, 18 + i * 12 + 10   # 每环宽度 10%、间隔 2%
            pie.add(
                dept,
                [(dept, v), ("占位", round(placeholder, 1))],
                radius=[f"{inner}%", f"{outer}%"],
                label_opts=opts.LabelOpts(formatter=LABEL_ONLY_VALUE, font_size=11),
            )
        pie.set_colors(PALETTE + ["#EBEBEB"] * 6)    # 占位段浅灰
        pie.set_global_opts(
            title_opts=opts.TitleOpts(
                title=f"{m}\n公司总人数为 {total}",
                pos_left="center", pos_top="middle",
                title_textstyle_opts=opts.TextStyleOpts(font_size=18, color="#333")),
            legend_opts=opts.LegendOpts(is_show=False),
            tooltip_opts=opts.TooltipOpts(formatter=JsCode(
                "function(p){return p.name === '占位' ? '' : p.seriesName + '：' + p.value + '人';}")),
        )
        tl.add(pie, m)
    tl.add_schema(play_interval=1800, is_auto_play=False, is_loop_play=False, pos_bottom="0%")
    return tl


if __name__ == "__main__":
    save_html(build_chart(), "02_动态跑道图")
    snapshot("02_动态跑道图", height=780)
```


## 03_动态南丁格尔圆环图.py — 03_动态南丁格尔圆环图

```python
"""03 动态南丁格尔圆环图：流量来源占比（Pie 玫瑰图 + Timeline 时间轴）

Excel 原效果：圆环图 + 选项按钮切换"近7天 / 近30天"。
Python 复现：rosetype="radius" 玫瑰图按占比映射半径，
            内半径 20% 形成圆环，Timeline 切换统计周期。
数据：各流量来源占比（单位：%）
"""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common import PALETTE, save_html, snapshot

from pyecharts import options as opts
from pyecharts.charts import Pie, Timeline

SOURCES = ["首页推荐", "关注页面", "搜索", "个人主页", "其他来源"]
SHARE = {
    "近7天": [0.44, 0.17, 0.15, 0.08, 0.16],
    "近30天": [0.38, 0.21, 0.15, 0.09, 0.17],
}


def build_chart():
    tl = Timeline(init_opts=opts.InitOpts(width="920px", height="560px"))
    for period, values in SHARE.items():
        data = [(s, round(v, 4)) for s, v in zip(SOURCES, values)]
        rose = (
            Pie(init_opts=opts.InitOpts(width="920px", height="560px"))
            .add(
                period, data,
                radius=["20%", "70%"],            # 内半径形成圆环
                center=["50%", "52%"],
                rosetype="radius",                # 南丁格尔玫瑰图：半径映射数值
                label_opts=opts.LabelOpts(formatter="{b}\n{d}%", font_size=13),
            )
            .set_colors(PALETTE)
            .set_global_opts(
                title_opts=opts.TitleOpts(title=f"流量来源占比（{period}）"),
                legend_opts=opts.LegendOpts(orient="vertical", pos_left="right", pos_top="middle"),
                tooltip_opts=opts.TooltipOpts(formatter="{b}: {c} ({d}%)"),
            )
        )
        tl.add(rose, period)
    tl.add_schema(play_interval=2000, is_auto_play=False, is_loop_play=False, pos_bottom="0%")
    return tl


if __name__ == "__main__":
    save_html(build_chart(), "03_动态南丁格尔圆环图")
    snapshot("03_动态南丁格尔圆环图")
```


## 04_动态组合图.py — 04_动态组合图

```python
"""04 动态组合图：销售额 / 利润 / 利润率（Bar + Line 双轴，图例控制显隐）

Excel 原效果：柱形图 + 折线图双轴呈现，复选框控制系列显隐。
Python 复现：Bar（销售额、利润）+ extend_axis 扩展第二 Y 轴 +
            Line（利润率，yaxis_index=1），点击图例即可显示/隐藏系列。
数据：2022 年化妆品销售情况
"""
import os
import sys

from pyecharts import options as opts
from pyecharts.charts import Bar, Line
from pyecharts.commons.utils import JsCode

CATEGORIES = ["口红", "面膜", "隔离", "粉底液"]

PROFIT = [1896, 1563, 986, 1324]
MARGIN = [0.6771, 0.6096, 0.4321, 0.5627]


def build_chart():
    bar = (
        Bar(init_opts=opts.InitOpts(width="920px", height="540px"))

        .extend_axis(
            yaxis=opts.AxisOpts(
                name="利润率", min_=0, max_=1,
                axislabel_opts=opts.LabelOpts(formatter=JsCode(

            )

            title_opts=opts.TitleOpts(title="2022年化妆品销售情况",

            legend_opts=opts.LegendOpts(pos_top="12%"),
```

## 05_数据透视表切片器.py — 05_数据透视表切片器

```python
"""05 数据透视表 + 切片器：各部门各学历平均月收入（pandas 透视 + Tab 选项卡）

Excel 原效果：数据透视表 + 部门切片器，点击部门即切换统计范围。
Python 复现：pandas pivot_table 复现透视计算，
            pyecharts Tab 选项卡等价于切片器，点击部门标签即切换图表。
数据：data/案例5_人力资源明细.csv（1470 名员工明细）
"""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common import DATA_DIR, save_html, snapshot

import pandas as pd
from pyecharts import options as opts
from pyecharts.charts import Bar, Tab

EDU_ORDER = ["高中", "专科", "本科", "硕士", "博士"]
CSV_PATH = os.path.join(DATA_DIR, "案例5_人力资源明细.csv")


def build_chart():
    df = pd.read_csv(CSV_PATH)
    tab = Tab()
    dept_list = ["全部"] + sorted(df["部门"].unique().tolist())

    for dept in dept_list:
        sub = df if dept == "全部" else df[df["部门"] == dept]
        g = sub.groupby("学历")["月收入"].mean().reindex(EDU_ORDER).round(0)
        bar = (
            Bar(init_opts=opts.InitOpts(width="920px", height="500px"))
            .add_xaxis(EDU_ORDER)
            .add_yaxis("平均月收入", g.tolist(), color="#4472C4",
                       label_opts=opts.LabelOpts(position="top", formatter="{c} 元"))
            .set_global_opts(
                title_opts=opts.TitleOpts(title=f"各学历平均月收入（{dept}，{len(sub)} 人）"),
                yaxis_opts=opts.AxisOpts(name="平均月收入（元）"),
                legend_opts=opts.LegendOpts(is_show=False),
            )
        )
        tab.add(bar, dept)   # 每个选项卡 ≈ 切片器的一个选项
    return tab


if __name__ == "__main__":
    save_html(build_chart(), "05_数据透视表切片器")
    snapshot("05_数据透视表切片器")
```


## 06_VBA动态玉玦图.py — 06_VBA动态玉玦图

```python
"""06 VBA 动态玉玦图：流量来源占比（多环 Pie + Tab 选项卡替代 VBA 按钮）

Excel 原效果：VBA 按钮切换"近7天 / 近30天"，玉玦（缺角圆环）逐级展开。
Python 复现：每个来源一个缺角圆环（数值段 + 占位段，占位 = 0.5 − 占比），

from pyecharts.charts import Pie, Tab
from pyecharts.commons.utils import JsCode

SOURCES = ["个人主页", "搜索", "关注页面", "首页推荐"]
SHARE = {
    "近7天": [0.13, 0.19, 0.32, 0.36],
    "近30天": [0.09, 0.24, 0.29, 0.38],
}

    "function(p){return p.name === '占位' ? '' : p.name + '\\n' + (p.value * 100).toFixed(0) + '%';}")
```

## 07_动态滑珠图.py — 07_动态滑珠图

```python
"""07 动态滑珠图：各区域目标完成率（堆积 Bar 滑轨 + EffectScatter 滑珠 + Timeline）

Excel 原效果：条形滑轨 + 滑珠，滚动条切换月份。
Python 复现：堆积柱形作浅色滑轨（完成率 + 未完成率 = 100%），
            EffectScatter 涟漪散点作滑珠，Timeline 切换月份。
数据：各区域月度目标完成率（单位：%）
"""
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common import save_html, snapshot

from pyecharts import options as opts
from pyecharts.charts import Bar, EffectScatter, Timeline
from pyecharts.commons.utils import JsCode

MONTHS = ["1月", "2月", "3月", "4月", "5月", "6月"]
REGIONS = ["华北", "华南", "东北", "西北", "西南", "华东"]
RATES = {
    "1月": [0.32, 0.49, 0.65, 0.73, 0.536, 0.32],
    "2月": [0.45, 0.36, 0.53, 0.63, 0.498, 0.49],
    "3月": [0.66, 0.54, 0.58, 0.61, 0.527, 0.65],
    "4月": [0.52, 0.39, 0.48, 0.53, 0.708, 0.73],
    "5月": [0.69, 0.415, 0.445, 0.47, 0.7035, 0.536],
    "6月": [0.771, 0.403, 0.399, 0.408, 0.758, 0.7468],
}


def build_chart():
    tl = Timeline(init_opts=opts.InitOpts(width="920px", height="540px"))
    for m in MONTHS:
        rates = [round(v, 4) for v in RATES[m]]
        bar = (
            Bar(init_opts=opts.InitOpts(width="920px", height="540px"))
            .add_xaxis(REGIONS)
            .add_yaxis("完成率", rates, stack="slider", color="#4472C4", bar_width="45%",
                       label_opts=opts.LabelOpts(is_show=False))
            .add_yaxis("未完成率", [round(1 - r, 4) for r in rates], stack="slider",
                       color="#DCE6F1", bar_width="45%",     # 浅色滑轨
                       label_opts=opts.LabelOpts(is_show=False))
            .set_global_opts(
                title_opts=opts.TitleOpts(title=f"{m}各区域目标完成率", subtitle="滑珠位置即完成率水平"),
                legend_opts=opts.LegendOpts(is_show=False),
                xaxis_opts=opts.AxisOpts(name="区域"),
                yaxis_opts=opts.AxisOpts(name="完成率", max_=1,
                                         axislabel_opts=opts.LabelOpts(formatter=JsCode(
                                             "function(v){return (v*100).toFixed(0)+'%';}"))),
                tooltip_opts=opts.TooltipOpts(trigger="axis", formatter=JsCode(
                    "function(ps){var p=ps[0];var v=Array.isArray(p.value)?p.value[1]:p.value;"
                    "return p.name+'：完成率 '+(v*100).toFixed(1)+'%';}")),
            )
        )
        bead = (
            EffectScatter()
            .add_xaxis(REGIONS)
            .add_yaxis("滑珠", rates, symbol_size=16, color="#ED7D31",
                       label_opts=opts.LabelOpts(is_show=False))
        )
        bar.overlap(bead)
        tl.add(bar, m)
    tl.add_schema(play_interval=1500, is_auto_play=False, is_loop_play=False, pos_bottom="0%")
    return tl


if __name__ == "__main__":
    save_html(build_chart(), "07_动态滑珠图")
    snapshot("07_动态滑珠图")
```

```python
"""08 综合数据大屏：Page 可拖拽布局整合全部案例

Excel 原效果：数据大屏将多个动态图表放在同一画布展示。
Python 复现：pyecharts Page(DraggablePageLayout) 整合案例 1—7，
            浏览器打开后可自由拖拽每个图表的位置与大小。
注：pyecharts 2.x 的 Page 不能嵌套 Tab 组件，
    因此大屏中案例 5 / 案例 6 各取一张代表性图表（全部部门、近 30 天）。
"""
import importlib
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from common import DATA_DIR, PALETTE, save_html, snapshot

import pandas as pd
from pyecharts import options as opts
from pyecharts.charts import Bar, Page, Pie
from pyecharts.commons.utils import JsCode

EDU_ORDER = ["高中", "专科", "本科", "硕士", "博士"]
SOURCES6 = ["个人主页", "搜索", "关注页面", "首页推荐"]
SHARE6_30 = [0.09, 0.24, 0.29, 0.38]

LABEL_PCT = JsCode(
    "function(p){return p.name === '占位' ? '' : p.name + '\\n' + (p.value * 100).toFixed(0) + '%';}")


def _build_bar5_all():
    """案例 5 代表性图表：全部员工各学历平均月收入。"""
    df = pd.read_csv(os.path.join(DATA_DIR, "案例5_人力资源明细.csv"))
    g = df.groupby("学历")["月收入"].mean().reindex(EDU_ORDER).round(0)
    return (
        Bar(init_opts=opts.InitOpts(width="920px", height="500px"))
        .add_xaxis(EDU_ORDER)
        .add_yaxis("平均月收入", g.tolist(), color="#4472C4",
                   label_opts=opts.LabelOpts(position="top", formatter="{c} 元"))
        .set_global_opts(
            title_opts=opts.TitleOpts(title=f"各学历平均月收入（全部，{len(df)} 人）"),
            yaxis_opts=opts.AxisOpts(name="平均月收入（元）"),
            legend_opts=opts.LegendOpts(is_show=False))
    )


def _build_pie6_30():
    """案例 6 代表性图表：近 30 天流量来源玉玦图。"""
    pie = Pie(init_opts=opts.InitOpts(width="920px", height="560px"))
    for i, (s, v) in enumerate(zip(SOURCES6, SHARE6_30)):
        inner, outer = 15 + i * 14, 15 + i * 14 + 11
        pie.add(s, [(s, v), ("占位", round(0.5 - v, 4))],
                radius=[f"{inner}%", f"{outer}%"],
                label_opts=opts.LabelOpts(formatter=LABEL_PCT, font_size=12))
    pie.set_colors(["#ED7D31", "#A5A5A5", "#FFC000", "#4472C4"] + ["#EBEBEB"] * 4)
    pie.set_global_opts(
        title_opts=opts.TitleOpts(title="流量来源玉玦图（近30天）", pos_top="2%"),
        legend_opts=opts.LegendOpts(is_show=False))
    return pie


def build_chart():
    tl1 = importlib.import_module("01_动态柱形图").build_chart()
    tl2 = importlib.import_module("02_动态跑道图").build_chart()
    tl3 = importlib.import_module("03_动态南丁格尔圆环图").build_chart()
    combo4 = importlib.import_module("04_动态组合图").build_chart()
    tl7 = importlib.import_module("07_动态滑珠图").build_chart()

    page = Page(page_title="第三章 动态图表 · Python 数据大屏",
                layout=Page.DraggablePageLayout)
    page.add(tl1, tl2, tl3, combo4, _build_bar5_all(), _build_pie6_30(), tl7)
    return page


if __name__ == "__main__":
    save_html(build_chart(), "08_综合数据大屏")
    print("提示：浏览器打开 output/08_综合数据大屏.html 后可自由拖拽布局")
    snapshot("08_综合数据大屏", height=1000, wait=4000)
```
